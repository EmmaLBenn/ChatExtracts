cleanlabels <- function(x, qname=FALSE) {
  is_factor <- FALSE
  if(is.factor(x)) {
    is_factor <- TRUE
    vec <- x
    x <- levels(x)
  }
  
  x <- x |> 
    str_replace("MINT_", ifelse(qname, "MINT - ", "")) |> 
    str_replace("_", " - ") 
  
  if(is_factor) {
    levels(vec) <- x
    return(vec)
  }
  x
}
